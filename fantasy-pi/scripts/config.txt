# Raspberry Pi Boot Configuration for Fantasy Pi
# Place this file in the root of the SD card boot partition

# Enable 64-bit mode (required for Fantasy Pi kernel)
arm_64bit=1

# Disable overscan for clean display output
disable_overscan=1

# Force HDMI hotplug (works without connected monitor at boot)
hdmi_force_hotplug=1

# HDMI mode: 640x480 @ 60Hz (safe default)
# Change to your preferred resolution:
#   hdmi_group=2, hdmi_mode=4   -> 640x480
#   hdmi_group=2, hdmi_mode=14  -> 848x480
#   hdmi_group=2, hdmi_mode=16  -> 1024x768
#   hdmi_group=2, hdmi_mode=23  -> 1280x768
#   hdmi_group=2, hdmi_mode=82  -> 1920x1080
hdmi_group=2
hdmi_mode=4

# GPU memory split (minimum needed for framebuffer)
gpu_mem=16

# Disable rainbow splash screen
disable_splash=1

# Enable UART for debug output
enable_uart=1
